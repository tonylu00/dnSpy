using System;
using System.Collections.Generic;
using System.IO;
using System.Linq;
using System.Globalization;
using dnlib.DotNet;
using dnlib.DotNet.Emit;
using dnlib.DotNet.Writer;
using DnSpyMetadataBamlSupport.dnSpy.BamlDecompiler.Baml;

public sealed class DnSpyRestoreMetadataFields : Microsoft.Build.Utilities.Task {
    [Microsoft.Build.Framework.Required]
    public string AssemblyPath { get; set; }
    public Microsoft.Build.Framework.ITaskItem[] ReferencePaths { get; set; }
    public bool TupleNamesOnly { get; set; }
    public bool PublicSign { get; set; }
    public override bool Execute() {
        const string prefix = "dnSpy.metadata.string-field.v1|";
        string temporary = AssemblyPath + ".metadata-" + Guid.NewGuid().ToString("N");
        string pdb = Path.ChangeExtension(AssemblyPath, ".pdb");
        string temporaryPdb = temporary + ".pdb";
        try {
            bool changed = false;
            using (var module = ModuleDefMD.Load(AssemblyPath, new ModuleCreationOptions { TryToLoadPdbFromDisk = true })) {
                module.LoadEverything();
                changed |= RestoreTupleNames(module);
                if (!TupleNamesOnly) {
                changed |= RestoreAccessorContracts(module);
                changed |= RestorePropertyNames(module);
                changed |= RestoreDelegateMethods(module);
                changed |= RestoreProtectedCalls(module);
                changed |= RestoreMemberNames(module, true);
                changed |= RestoreMemberNames(module, false);
                changed |= RestoreTypeNames(module);
                changed |= RestoreDefinitionFlags(module);
                changed |= RestoreStaticValueTypes(module);
                changed |= RestoreUncheckedReferences(module);
                changed |= RestoreParameterNames(module);
                changed |= RestoreFieldData(module);
                changed |= RestoreEntryPoint(module);
                var rawTypes = module.GetTypes().Where(t => t.CustomAttributes.Any(a => a.TypeFullName == "System.Reflection.ObfuscationAttribute" &&
                    a.NamedArguments.Any(n => n.Name == "Feature" && n.Value != null && n.Value.ToString() == "dnSpy.metadata.raw-exceptions.v1"))).ToArray();
                foreach (var rawType in rawTypes) {
                    var throwHelper = rawType.Methods.SingleOrDefault(m => m.Name == "ThrowValue" && m.IsStatic &&
                        m.MethodSig != null && m.MethodSig.Params.Count == 1 && m.MethodSig.Params[0].FullName == "System.Object" &&
                        m.MethodSig.RetType.FullName == "System.Exception");
                    if (rawType.BaseType == null || rawType.BaseType.FullName != "System.Exception" || throwHelper == null)
                        throw new InvalidDataException("Invalid raw exception placeholder.");
                    foreach (var method in module.GetTypes().Where(t => t != rawType).SelectMany(t => t.Methods)) {
                        if (!method.HasBody) continue;
                        foreach (var handler in method.Body.ExceptionHandlers)
                            if (handler.CatchType == rawType) handler.CatchType = module.CorLibTypes.Object.TypeDefOrRef;
                        foreach (var local in method.Body.Variables)
                            if (local.Type.ToTypeDefOrRef() == rawType) local.Type = module.CorLibTypes.Object;
                        foreach (var instruction in method.Body.Instructions) {
                            var called = instruction.Operand as IMethod;
                            if (instruction.OpCode == OpCodes.Call && called != null && called.DeclaringType == rawType && called.ResolveMethodDef() == throwHelper) {
                                instruction.OpCode = OpCodes.Nop;
                                instruction.Operand = null;
                            }
                            else if (instruction.Operand == rawType) instruction.Operand = module.CorLibTypes.Object.TypeDefOrRef;
                        }
                    }
                    module.Types.Remove(rawType);
                    changed = true;
                }
                foreach (var type in module.GetTypes()) {
                    foreach (var attribute in type.CustomAttributes.ToArray()) {
                        if (attribute.TypeFullName != "System.Reflection.ObfuscationAttribute") continue;
                        var feature = attribute.NamedArguments.FirstOrDefault(a => a.Name == "Feature");
                        string text = feature == null || feature.Value == null ? null : feature.Value.ToString();
                        const string usagePrefix = "dnSpy.metadata.attribute-usage.v1|";
                        if (text != null && text.StartsWith(usagePrefix, StringComparison.Ordinal)) {
                            var usage = type.CustomAttributes.Single(a => a.TypeFullName == "System.AttributeUsageAttribute");
                            var argument = usage.ConstructorArguments[0];
                            usage.ConstructorArguments[0] = new CAArgument(argument.Type,
                                int.Parse(text.Substring(usagePrefix.Length), CultureInfo.InvariantCulture));
                            type.CustomAttributes.Remove(attribute);
                            changed = true;
                            continue;
                        }
                        if (text == null || !text.StartsWith(prefix, StringComparison.Ordinal)) continue;
                        var parts = text.Substring(prefix.Length).Split('|');
                        if (parts.Length != 4) throw new InvalidDataException("Invalid dnSpy field metadata instruction.");
                        var name = new UTF8String(Convert.FromBase64String(parts[0]));
                        var field = new FieldDefUser(name, new FieldSig(module.CorLibTypes.String),
                            (dnlib.DotNet.FieldAttributes)int.Parse(parts[1], CultureInfo.InvariantCulture));
                        if (parts[2] != "none") {
                            var kind = (ElementType)int.Parse(parts[2], CultureInfo.InvariantCulture);
                            string value = null;
                            if (parts[3] != "N") {
                                if (!parts[3].StartsWith("V", StringComparison.Ordinal)) throw new InvalidDataException("Invalid constant encoding.");
                                var bytes = Convert.FromBase64String(parts[3].Substring(1));
                                if (bytes.Length % 2 != 0) throw new InvalidDataException("Invalid UTF-16 constant.");
                                value = new string(Enumerable.Range(0, bytes.Length / 2).Select(i => (char)(bytes[i * 2] | bytes[i * 2 + 1] << 8)).ToArray());
                            }
                            object constant;
                            if (kind == ElementType.I1) constant = sbyte.Parse(value, CultureInfo.InvariantCulture);
                            else if (kind == ElementType.String) constant = value;
                            else if (kind == ElementType.Class) constant = null;
                            else throw new InvalidDataException("Unsupported metadata constant type.");
                            field.Constant = new ConstantUser(constant, kind);
                        }
                        type.Fields.Add(field);
                        type.CustomAttributes.Remove(attribute);
                        changed = true;
                    }
                }
                }
                if (!TupleNamesOnly) changed |= RestoreGlobalMembers(module);
                if (!changed) return true;
                if (module.IsStrongNameSigned && !PublicSign) throw new InvalidOperationException("Metadata restoration requires an unsigned output or explicit public signing.");
                var options = new ModuleWriterOptions(module) {
                    WritePdb = module.PdbState != null, PdbFileName = temporaryPdb,
                    PdbFileNameInDebugDirectory = Path.GetFileName(pdb)
                };
                if (PublicSign && module.Assembly?.PublicKey?.Data?.Length > 0) {
                    options.StrongNamePublicKey = new StrongNamePublicKey(module.Assembly.PublicKey);
                    options.Cor20HeaderOptions.Flags = module.Cor20HeaderFlags | dnlib.DotNet.MD.ComImageFlags.StrongNameSigned;
                }
                module.Write(temporary, options);
            }
            File.Copy(temporary, AssemblyPath, true);
            if (File.Exists(temporaryPdb)) File.Copy(temporaryPdb, pdb, true);
            return true;
        }
        catch (Exception error) { Log.LogErrorFromException(error); return false; }
        finally {
            if (File.Exists(temporary)) File.Delete(temporary);
            if (File.Exists(temporaryPdb)) File.Delete(temporaryPdb);
        }
    }

    static bool RestoreProtectedCalls(ModuleDef module) {
        const string prefix = "dnSpy.metadata.protected-call.v1|";
        var maps = new Dictionary<MethodDef, MethodDef>();
        foreach (var type in module.GetTypes()) foreach (var stub in type.Methods) {
            foreach (var attribute in stub.CustomAttributes) {
                if (attribute.TypeFullName != "System.Reflection.ObfuscationAttribute") continue;
                var feature = attribute.NamedArguments.FirstOrDefault(a => a.Name == "Feature");
                string text = feature == null || feature.Value == null ? null : feature.Value.ToString();
                if (text == null || !text.StartsWith(prefix, StringComparison.Ordinal)) continue;
                var name = new UTF8String(Convert.FromBase64String(text.Substring(prefix.Length)));
                var target = type.Methods.SingleOrDefault(m => m != stub && m.Name == name && new SigComparer().Equals(m.MethodSig, stub.MethodSig));
                if (target == null || !target.IsFamily || target.IsVirtual || target.IsStatic || !stub.IsFamily || stub.IsStatic || stub.IsVirtual)
                    throw new InvalidDataException("Invalid protected call compilation stub.");
                maps.Add(stub, target);
            }
        }
        if (maps.Count == 0) return false;
        var byName = maps.Keys.ToDictionary(m => m.Name.String, StringComparer.Ordinal);
        var methods = module.GetTypes().SelectMany(t => t.Methods).ToArray();
        var references = new HashSet<MemberRef>(module.GetMemberRefs());
        foreach (var method in methods.Where(m => m.HasBody)) foreach (var instruction in method.Body.Instructions) {
            var called = instruction.Operand as IMethod;
            var specification = called as MethodSpec;
            var reference = (specification == null ? called : specification.Method) as MemberRef;
            if (reference != null) references.Add(reference);
            var definition = (specification == null ? called : specification.Method) as MethodDef;
            MethodDef target;
            if (definition != null && maps.TryGetValue(definition, out target)) {
                if (specification != null) specification.Method = target; else instruction.Operand = target;
            }
        }
        foreach (var reference in references) {
            MethodDef stub;
            if (!reference.IsMethodRef || !byName.TryGetValue(reference.Name, out stub)) continue;
            var owner = reference.DeclaringType.ResolveTypeDef();
            // These stubs are module-local. Never reinterpret an unrelated
            // external member merely because its name matches a local stub.
            if (owner == null || owner.Module != module) continue;
            if (!new SigComparer().Equals(reference.MethodSig, stub.MethodSig)) throw new InvalidDataException("Protected stub signature mismatch.");
            var target = maps[stub];
            if (owner != target.DeclaringType) {
                if (target.DeclaringType.HasGenericParameters) throw new InvalidDataException("Protected stub generic owner requires an exact declaring-type reference.");
                reference.Class = target.DeclaringType;
            }
            reference.Name = target.Name;
        }
        foreach (var stub in maps.Keys) stub.DeclaringType.Methods.Remove(stub);
        return true;
    }

    static bool RestoreEntryPoint(ModuleDef module) {
        Func<CustomAttribute, string, bool> marker = (a, feature) => a.TypeFullName == "System.Reflection.ObfuscationAttribute" &&
            a.NamedArguments.Any(n => n.Name == "Feature" && n.Value != null && n.Value.ToString() == feature);
        var stubs = module.Types.Where(t => t.CustomAttributes.Any(a => marker(a, "dnSpy.metadata.entry-point-stub.v1"))).ToArray();
        if (stubs.Length == 0) return false;
        var methods = module.GetTypes().SelectMany(t => t.Methods).Where(m => m.CustomAttributes.Any(a => marker(a, "dnSpy.metadata.entry-point.v1"))).ToArray();
        if (stubs.Length != 1 || methods.Length != 1 || !methods[0].IsStatic || !methods[0].HasBody || module.EntryPoint?.DeclaringType != stubs[0])
            throw new InvalidDataException("Invalid original entry-point instruction or missing compilation stub.");
        var entry = methods[0];
        foreach (var attribute in entry.CustomAttributes.Where(a => marker(a, "dnSpy.metadata.entry-point.v1")).ToArray()) entry.CustomAttributes.Remove(attribute);
        module.EntryPoint = entry;
        if (module.PdbState != null) module.PdbState.UserEntryPoint = entry;
        module.Types.Remove(stubs[0]);
        return true;
    }

    static bool RestoreTupleNames(ModuleDef module) {
        const string prefix = "dnSpy.metadata.tuple-names.v1|";
        const string name = "System.Runtime.CompilerServices.TupleElementNamesAttribute";
        bool changed = false;
        var owners = new List<IHasCustomAttribute>();
        foreach (var type in module.GetTypes()) {
            owners.Add(type); owners.AddRange(type.Fields); owners.AddRange(type.Properties); owners.AddRange(type.Events);
            foreach (var method in type.Methods) { owners.Add(method); owners.AddRange(method.ParamDefs); }
        }
        foreach (var owner in owners) foreach (var marker in owner.CustomAttributes.ToArray()) {
            if (marker.TypeFullName != "System.Reflection.ObfuscationAttribute") continue;
            string feature = marker.NamedArguments.Where(n => n.Name == "Feature").Select(n => n.Value == null ? null : n.Value.ToString()).FirstOrDefault();
            if (feature == null || !feature.StartsWith(prefix, StringComparison.Ordinal)) continue;
            var parts = feature.Substring(prefix.Length).Split('|');
            if (parts.Length != 3) throw new InvalidDataException("Invalid tuple-name instruction.");
            IHasCustomAttribute target = owner;
            if (parts[0] != "s") {
                MethodDef method; string sequence;
                if (parts[0].StartsWith("p", StringComparison.Ordinal)) { method = owner as MethodDef; sequence = parts[0].Substring(1); }
                else if (parts[0].StartsWith("d", StringComparison.Ordinal) && owner is TypeDef delegateType) {
                    var split = parts[0].Substring(1).Split(':');
                    if (split.Length != 2) throw new InvalidDataException("Invalid delegate tuple target.");
                    var methodName = new UTF8String(Convert.FromBase64String(split[0]));
                    method = delegateType.Methods.Single(m => m.Name == methodName); sequence = split[1];
                }
                else throw new InvalidDataException("Invalid tuple target.");
                ushort index = ushort.Parse(sequence, CultureInfo.InvariantCulture);
                if (method == null || index > method.MethodSig.Params.Count) throw new InvalidDataException("Invalid tuple parameter position.");
                var parameter = method.ParamDefs.SingleOrDefault(p => p.Sequence == index);
                if (parameter == null) { parameter = new ParamDefUser("", index); method.ParamDefs.Add(parameter); }
                target = parameter;
            }
            var names = new List<CAArgument>();
            foreach (var item in parts[2].Length == 0 ? new string[0] : parts[2].Split(',')) {
                string value = null;
                if (item != "N") {
                    if (!item.StartsWith("V", StringComparison.Ordinal)) throw new InvalidDataException("Invalid tuple element name.");
                    var bytes = Convert.FromBase64String(item.Substring(1));
                    if (bytes.Length % 2 != 0) throw new InvalidDataException("Invalid tuple UTF-16 name.");
                    value = new string(Enumerable.Range(0, bytes.Length / 2).Select(i => (char)(bytes[i * 2] | bytes[i * 2 + 1] << 8)).ToArray());
                }
                names.Add(new CAArgument(module.CorLibTypes.String, value == null ? null : new UTF8String(value)));
            }
            var existing = target.CustomAttributes.Where(a => a.TypeFullName == name).ToArray();
            if (existing.Length != 0) {
                var actual = existing.Length == 1 && existing[0].ConstructorArguments.Count == 1 ? existing[0].ConstructorArguments[0].Value as IList<CAArgument> : null;
                if (actual == null || !actual.Select(a => a.Value == null ? null : a.Value.ToString()).SequenceEqual(names.Select(a => a.Value == null ? null : a.Value.ToString())))
                    throw new InvalidDataException("Conflicting tuple element names.");
            }
            else {
                var assembly = new AssemblyRefUser(new AssemblyNameInfo(System.Text.Encoding.UTF8.GetString(Convert.FromBase64String(parts[1]))));
                ITypeDefOrRef attributeType;
                if (assembly.Name == module.Assembly.Name && assembly.Version == module.Assembly.Version && assembly.Culture == module.Assembly.Culture) {
                    attributeType = module.Find(name, false);
                    if (attributeType == null) throw new InvalidDataException("Local tuple attribute type was not retained.");
                }
                else attributeType = new TypeRefUser(module, "System.Runtime.CompilerServices", "TupleElementNamesAttribute", assembly);
                var constructor = new MemberRefUser(module, ".ctor", MethodSig.CreateInstance(module.CorLibTypes.Void, new SZArraySig(module.CorLibTypes.String)), attributeType);
                var attribute = new CustomAttribute(constructor);
                attribute.ConstructorArguments.Add(new CAArgument(new SZArraySig(module.CorLibTypes.String), names));
                target.CustomAttributes.Add(attribute);
            }
            owner.CustomAttributes.Remove(marker); changed = true;
        }
        return changed;
    }

    static bool RestoreFieldData(ModuleDef module) {
        const string prefix = "dnSpy.metadata.field-data.v1|";
        bool changed = false;
        foreach (var field in module.GetTypes().SelectMany(t => t.Fields)) {
            int count = 0;
            foreach (var attribute in field.CustomAttributes.ToArray()) {
                if (attribute.TypeFullName != "System.Reflection.ObfuscationAttribute") continue;
                string feature = attribute.NamedArguments.Where(n => n.Name == "Feature").Select(n => n.Value == null ? null : n.Value.ToString()).FirstOrDefault();
                if (feature == null || !feature.StartsWith(prefix, StringComparison.Ordinal)) continue;
                var parts = feature.Substring(prefix.Length).Split('|');
                if (++count != 1 || parts.Length != 5 || field.Name != new UTF8String(Convert.FromBase64String(parts[0])))
                    throw new InvalidDataException("Invalid field data instruction or unrestored field name.");
                object value = null;
                var kind = parts[2] == "none" ? ElementType.End : (ElementType)int.Parse(parts[2], CultureInfo.InvariantCulture);
                if (parts[3] != "N") {
                    if (!parts[3].StartsWith("V", StringComparison.Ordinal)) throw new InvalidDataException("Invalid field constant encoding.");
                    using (var stream = new MemoryStream(Convert.FromBase64String(parts[3].Substring(1))))
                    using (var reader = new BinaryReader(stream)) {
                        switch (kind) {
                        case ElementType.Boolean: value = reader.ReadBoolean(); break;
                        case ElementType.Char: value = (char)reader.ReadUInt16(); break;
                        case ElementType.I1: value = reader.ReadSByte(); break;
                        case ElementType.U1: value = reader.ReadByte(); break;
                        case ElementType.I2: value = reader.ReadInt16(); break;
                        case ElementType.U2: value = reader.ReadUInt16(); break;
                        case ElementType.I4: value = reader.ReadInt32(); break;
                        case ElementType.U4: value = reader.ReadUInt32(); break;
                        case ElementType.I8: value = reader.ReadInt64(); break;
                        case ElementType.U8: value = reader.ReadUInt64(); break;
                        case ElementType.R4: value = reader.ReadSingle(); break;
                        case ElementType.R8: value = reader.ReadDouble(); break;
                        case ElementType.String:
                            if (stream.Length % 2 != 0) throw new InvalidDataException("Invalid UTF-16 field constant.");
                            var chars = new char[stream.Length / 2];
                            for (int i = 0; i < chars.Length; i++) chars[i] = (char)reader.ReadUInt16();
                            value = new string(chars); break;
                        default: throw new InvalidDataException("Unsupported field constant kind.");
                        }
                        if (stream.Position != stream.Length) throw new InvalidDataException("Trailing field constant data.");
                    }
                }
                field.Constant = kind == ElementType.End ? null : new ConstantUser(value, kind);
                if (parts[4] != "none") field.InitialValue = Convert.FromBase64String(parts[4]);
                field.Attributes = (dnlib.DotNet.FieldAttributes)uint.Parse(parts[1], CultureInfo.InvariantCulture);
                field.CustomAttributes.Remove(attribute); changed = true;
            }
        }
        return changed;
    }

    static bool RestoreParameterNames(ModuleDef module) {
        const string prefix = "dnSpy.metadata.parameter-names.v1|";
        bool changed = false;
        var parameterOwners = module.GetTypes().SelectMany(t => t.Methods).SelectMany(m => m.ParamDefs.Select(p => new { Parameter = p, Method = m })).ToDictionary(p => p.Parameter, p => p.Method);
        foreach (var type in module.GetTypes()) {
            foreach (var owner in new IHasCustomAttribute[] { type }.Concat(type.Methods.Cast<IHasCustomAttribute>()).Concat(type.Methods.SelectMany(m => m.ParamDefs).Cast<IHasCustomAttribute>()).ToArray()) {
                var restored = new HashSet<string>(StringComparer.Ordinal);
                foreach (var attribute in owner.CustomAttributes.ToArray()) {
                    if (attribute.TypeFullName != "System.Reflection.ObfuscationAttribute") continue;
                    string feature = attribute.NamedArguments.Where(n => n.Name == "Feature").Select(n => n.Value == null ? null : n.Value.ToString()).FirstOrDefault();
                    if (feature == null || !feature.StartsWith(prefix, StringComparison.Ordinal)) continue;
                    var parts = feature.Substring(prefix.Length).Split('|');
                    if (parts.Length != 2) throw new InvalidDataException("Invalid parameter name instruction.");
                    MethodDef method = owner as MethodDef;
                    if (owner is ParamDef parameterOwner) method = parameterOwners[parameterOwner];
                    if (parts[0].StartsWith("d", StringComparison.Ordinal) && owner is TypeDef) {
                        var name = new UTF8String(Convert.FromBase64String(parts[0].Substring(1)));
                        method = type.Methods.Single(m => m.IsRuntime && m.Name == name);
                    }
                    else if (parts[0].StartsWith("c", StringComparison.Ordinal) && owner is TypeDef) {
                        var name = new UTF8String(Convert.FromBase64String(parts[0].Substring(1)));
                        method = type.Methods.Single(m => m.IsConstructor && m.Name == name && m.MethodSig.Params.Count == 0);
                    }
                    else if (parts[0] != "s") throw new InvalidDataException("Invalid parameter name target.");
                    foreach (var row in parts[1].Split(';')) {
                        var columns = row.Split(':');
                        if (columns.Length != 3 || !restored.Add(parts[0] + ":" + columns[0] + ":" + columns[1])) throw new InvalidDataException("Invalid or duplicate parameter name row.");
                        ushort index = ushort.Parse(columns[1], CultureInfo.InvariantCulture);
                        var name = new UTF8String(Convert.FromBase64String(columns[2]));
                        if (columns[0] == "g") {
                            var parameters = method == null ? type.GenericParameters : method.GenericParameters;
                            parameters.Single(p => p.Number == index).Name = name;
                        }
                        else if (columns[0] == "p" && method != null && index <= method.MethodSig.Params.Count) {
                            var parameter = method.ParamDefs.SingleOrDefault(p => p.Sequence == index);
                            if (parameter == null) { parameter = new ParamDefUser(name, index); method.ParamDefs.Add(parameter); }
                            else parameter.Name = name;
                        }
                        else throw new InvalidDataException("Invalid parameter name position.");
                    }
                    owner.CustomAttributes.Remove(attribute); changed = true;
                }
            }
        }
        return changed;
    }

    static bool RestoreDefinitionFlags(ModuleDef module) {
        bool changed = false;
        foreach (var type in module.GetTypes()) {
            foreach (var owner in new IHasCustomAttribute[] { type }.Concat(type.Methods.Cast<IHasCustomAttribute>())) {
                string prefix = "dnSpy.metadata." + (owner is TypeDef ? "type" : "method") + "-flags.v1|";
                int count = 0;
                foreach (var attribute in owner.CustomAttributes.ToArray()) {
                    if (attribute.TypeFullName != "System.Reflection.ObfuscationAttribute") continue;
                    string feature = attribute.NamedArguments.Where(n => n.Name == "Feature").Select(n => n.Value == null ? null : n.Value.ToString()).FirstOrDefault();
                    if (feature == null || !feature.StartsWith(prefix, StringComparison.Ordinal)) continue;
                    if (++count != 1) throw new InvalidDataException("Duplicate metadata flag instruction.");
                    uint flags = uint.Parse(feature.Substring(prefix.Length), CultureInfo.InvariantCulture);
                    if (owner is TypeDef definition) definition.Attributes = (TypeAttributes)flags;
                    else ((MethodDef)owner).Attributes = (MethodAttributes)flags;
                    owner.CustomAttributes.Remove(attribute); changed = true;
                }
            }
        }
        return changed;
    }

    static bool RestoreStaticValueTypes(ModuleDef module) {
        const string prefix = "dnSpy.metadata.static-values.v1|";
        const string methodFeature = "dnSpy.metadata.extension-method.v1";
        string Feature(CustomAttribute attribute) => attribute.TypeFullName != "System.Reflection.ObfuscationAttribute" ? null :
            attribute.NamedArguments.Where(n => n.Name == "Feature").Select(n => n.Value == null ? null : n.Value.ToString()).FirstOrDefault();
        void AddExtension(IHasCustomAttribute owner) {
            const string name = "System.Runtime.CompilerServices.ExtensionAttribute";
            if (owner.CustomAttributes.Any(a => a.TypeFullName == name)) return;
            var type = module.CorLibTypes.GetTypeRef("System.Runtime.CompilerServices", "ExtensionAttribute");
            owner.CustomAttributes.Add(new CustomAttribute(new MemberRefUser(module, ".ctor", MethodSig.CreateInstance(module.CorLibTypes.Void), type)));
        }
        bool changed = false;
        foreach (var type in module.GetTypes()) {
            var markers = type.CustomAttributes.Where(a => Feature(a)?.StartsWith(prefix, StringComparison.Ordinal) == true).ToArray();
            if (markers.Length == 0) continue;
            if (markers.Length != 1 || !type.IsAbstract || !type.IsSealed) throw new InvalidDataException("Invalid static value projection.");
            var flags = Feature(markers[0]).Substring(prefix.Length).Split('|');
            if (flags.Length != 2 || flags.Any(f => f != "0" && f != "1")) throw new InvalidDataException("Invalid extension flags.");
            var constructors = type.Methods.Where(m => m.IsInstanceConstructor).ToArray();
            if (constructors.Length != 1 || !constructors[0].IsInstanceConstructor || !constructors[0].IsPrivate || constructors[0].MethodSig.Params.Count != 0)
                throw new InvalidDataException("Missing or invalid temporary constructor.");
            type.Methods.Remove(constructors[0]);
            if (flags[0] == "1") AddExtension(type);
            if (flags[1] == "1") AddExtension(module.Assembly);
            foreach (var method in type.Methods) foreach (var attribute in method.CustomAttributes.Where(a => Feature(a) == methodFeature).ToArray()) {
                AddExtension(method); method.CustomAttributes.Remove(attribute);
            }
            type.CustomAttributes.Remove(markers[0]); changed = true;
        }
        return changed;
    }

    static bool RestoreGlobalMembers(ModuleDef module) {
        var companions = module.Types.Where(t => t.CustomAttributes.Any(a => a.TypeFullName == "System.Reflection.ObfuscationAttribute" &&
            a.NamedArguments.Any(n => n.Name == "Feature" && n.Value != null && n.Value.ToString() == "dnSpy.metadata.global-members.v1"))).ToArray();
        if (companions.Length == 0) return false;
        if (companions.Length != 1) throw new InvalidDataException("Multiple global member companions.");
        var source = companions[0]; var target = module.GlobalType;
        if (source == target || target == null || source.Methods.Any(m => !m.IsStatic) || source.Fields.Any(f => !f.IsStatic) || source.HasGenericParameters || source.HasProperties || source.HasEvents)
            throw new InvalidDataException("Invalid global member companion.");
        if (target.HasMethods || target.HasFields) throw new InvalidDataException("Global members already exist.");
        foreach (var reference in module.GetMemberRefs())
            if (reference.DeclaringType == source || reference.DeclaringType?.ResolveTypeDef() == source) reference.Class = target;
        foreach (var field in source.Fields.ToArray()) {
            foreach (var attribute in field.CustomAttributes.ToArray()) {
                if (attribute.TypeFullName != "System.Reflection.ObfuscationAttribute") continue;
                string feature = attribute.NamedArguments.Where(n => n.Name == "Feature").Select(n => n.Value == null ? null : n.Value.ToString()).FirstOrDefault();
                const string prefix = "dnSpy.metadata.global-field.v1|";
                if (feature == null || !feature.StartsWith(prefix, StringComparison.Ordinal)) continue;
                field.Attributes = (FieldAttributes)uint.Parse(feature.Substring(prefix.Length), CultureInfo.InvariantCulture);
                field.CustomAttributes.Remove(attribute);
            }
            source.Fields.Remove(field); target.Fields.Add(field);
        }
        foreach (var method in source.Methods.ToArray()) { source.Methods.Remove(method); target.Methods.Add(method); }
        foreach (var nested in source.NestedTypes.ToArray()) { source.NestedTypes.Remove(nested); target.NestedTypes.Add(nested); }
        module.Types.Remove(source);
        return true;
    }

    bool RestorePropertyNames(ModuleDef module) {
        const string prefix = "dnSpy.metadata.property-name.v1|";
        const string signaturePrefix = "dnSpy.metadata.property-has-this.v1|";
        bool changed = false;
        foreach (var property in module.GetTypes().SelectMany(t => t.Properties)) {
            int count = 0;
            foreach (var attribute in property.CustomAttributes.ToArray()) {
                if (attribute.TypeFullName != "System.Reflection.ObfuscationAttribute") continue;
                string feature = attribute.NamedArguments.Where(n => n.Name == "Feature").Select(n => n.Value == null ? null : n.Value.ToString()).FirstOrDefault();
                if (feature != null && feature.StartsWith(signaturePrefix, StringComparison.Ordinal)) {
                    string value = feature.Substring(signaturePrefix.Length);
                    if (value != "0" && value != "1") throw new InvalidDataException("Invalid property signature instruction.");
                    property.PropertySig.HasThis = value == "1";
                    property.CustomAttributes.Remove(attribute); changed = true; continue;
                }
                if (feature == null || !feature.StartsWith(prefix, StringComparison.Ordinal)) continue;
                if (++count != 1) throw new InvalidDataException("Duplicate property name instruction.");
                property.Name = new UTF8String(Convert.FromBase64String(feature.Substring(prefix.Length)));
                property.CustomAttributes.Remove(attribute); changed = true;
            }
        }
        return changed;
    }

    bool RestoreAccessorContracts(ModuleDef module) {
        const string prefix = "dnSpy.metadata.accessor-contract.v1|";
        var loaded = new List<ModuleDefMD>();
        var modules = new List<ModuleDef> { module };
        try {
            foreach (var path in (ReferencePaths ?? new Microsoft.Build.Framework.ITaskItem[0]).Select(p => p.ItemSpec).Distinct(StringComparer.OrdinalIgnoreCase)) {
                var reference = ModuleDefMD.Load(path);
                loaded.Add(reference); modules.Add(reference);
            }
            Func<IAssembly, string> identity = assembly => assembly == null ? "" : assembly.FullName;
            var assemblies = modules.ToLookup(m => identity(m.Assembly), StringComparer.Ordinal);
            var contracts = new Dictionary<MethodDef, Tuple<UTF8String, MethodAttributes, CustomAttribute>>();
            var inspected = new HashSet<MethodDef>();
            Action<MethodDef> inspect = method => {
                if (!inspected.Add(method)) return;
                foreach (var attribute in method.CustomAttributes) {
                    if (attribute.TypeFullName != "System.Reflection.ObfuscationAttribute") continue;
                    string feature = attribute.NamedArguments.Where(n => n.Name == "Feature").Select(n => n.Value == null ? null : n.Value.ToString()).FirstOrDefault();
                    if (feature == null || !feature.StartsWith(prefix, StringComparison.Ordinal)) continue;
                    var parts = feature.Substring(prefix.Length).Split('|');
                    if (parts.Length != 2 || contracts.ContainsKey(method)) throw new InvalidDataException("Invalid or duplicate accessor contract.");
                    contracts.Add(method, Tuple.Create(new UTF8String(Convert.FromBase64String(parts[0])),
                        (MethodAttributes)uint.Parse(parts[1], CultureInfo.InvariantCulture), attribute));
                }
            };
            foreach (var method in module.GetTypes().SelectMany(t => t.Methods)) inspect(method);
            var references = new HashSet<MemberRef>(module.GetMemberRefs().Where(r => r.IsMethodRef));
            foreach (var method in module.GetTypes().SelectMany(t => t.Methods))
                foreach (var implementation in method.Overrides) {
                    if (implementation.MethodDeclaration is MemberRef declaration) references.Add(declaration);
                    if (implementation.MethodBody is MemberRef body) references.Add(body);
                }
            foreach (var method in module.GetTypes().SelectMany(t => t.Methods).Where(m => m.HasBody))
                foreach (var instruction in method.Body.Instructions) {
                    var reference = instruction.Operand as MemberRef;
                    if (reference == null && instruction.Operand is MethodSpec specification) reference = specification.Method as MemberRef;
                    if (reference != null && reference.IsMethodRef) references.Add(reference);
                }
            var renames = new Dictionary<MemberRef, UTF8String>();
            foreach (var reference in references) {
                var owners = new Queue<ITypeDefOrRef>(); owners.Enqueue(reference.DeclaringType);
                var visited = new HashSet<string>(StringComparer.Ordinal);
                var matches = new List<MethodDef>();
                while (owners.Count != 0 && matches.Count == 0) {
                    var owner = owners.Dequeue();
                    if (owner is TypeSpec specification) {
                        var generic = specification.TypeSig.RemovePinnedAndModifiers() as GenericInstSig;
                        if (generic == null) continue;
                        owner = generic.GenericType.TypeDefOrRef;
                    }
                    string key = identity(owner.DefinitionAssembly) + "|" + owner.FullName;
                    if (!visited.Add(key)) continue;
                    foreach (var type in assemblies[identity(owner.DefinitionAssembly)].Select(m => m.Find(owner.FullName, false)).Where(t => t != null)) {
                        matches.AddRange(type.Methods.Where(m => m.Name == reference.Name && new SigComparer().Equals(m.MethodSig, reference.MethodSig)));
                        if (type.BaseType != null) owners.Enqueue(type.BaseType);
                    }
                }
                foreach (var method in matches) inspect(method);
                var values = matches.Where(contracts.ContainsKey).Select(m => contracts[m].Item1).Distinct().ToArray();
                if (values.Length > 1 || values.Length == 1 && matches.Any(m => !contracts.ContainsKey(m) && m.Name != values[0]))
                    throw new InvalidDataException("Conflicting accessor contracts in the selected compilation references: " + reference.FullName);
                if (values.Length == 1 && reference.Name != values[0]) renames.Add(reference, values[0]);
            }
            // Resolve all source signatures before changing any definition or reference.
            foreach (var entry in renames) entry.Key.Name = entry.Value;
            bool changed = renames.Count != 0;
            foreach (var entry in contracts.Where(p => p.Key.Module == module)) {
                entry.Key.Name = entry.Value.Item1;
                entry.Key.Attributes = entry.Value.Item2;
                entry.Key.CustomAttributes.Remove(entry.Value.Item3);
                changed = true;
            }
            return changed;
        }
        finally { foreach (var reference in loaded) reference.Dispose(); }
    }

    static bool RestoreUncheckedReferences(ModuleDef module) {
        var helpers = module.GetTypes().Where(t => t.CustomAttributes.Any(a => a.TypeFullName == "System.Reflection.ObfuscationAttribute" &&
            a.NamedArguments.Any(n => n.Name == "Feature" && n.Value != null && n.Value.ToString() == "dnSpy.metadata.unchecked-references.v1"))).ToArray();
        foreach (var helper in helpers) {
            var cast = helper.Methods.SingleOrDefault(m => m.Name == "Cast" && m.IsStatic && m.GenericParameters.Count == 1 &&
                m.MethodSig.Params.Count == 1 && m.MethodSig.Params[0].ElementType == ElementType.Object && m.MethodSig.RetType is GenericMVar);
            var asRef = helper.Methods.SingleOrDefault(m => m.Name == "AsRef" && m.IsStatic && m.GenericParameters.Count == 2 &&
                m.MethodSig.Params.Count == 1 && m.MethodSig.Params[0] is ByRefSig &&
                m.MethodSig.Params[0].Next is GenericMVar && ((GenericMVar)m.MethodSig.Params[0].Next).Number == 0 &&
                m.MethodSig.RetType is ByRefSig && m.MethodSig.RetType.Next is GenericMVar &&
                ((GenericMVar)m.MethodSig.RetType.Next).Number == 1);
            if (cast == null || helper.Methods.Count != (asRef == null ? 1 : 2) || helper.HasFields) throw new InvalidDataException("Invalid unchecked reference placeholder.");
            foreach (var method in module.GetTypes().Where(t => t != helper).SelectMany(t => t.Methods).Where(m => m.HasBody))
                foreach (var instruction in method.Body.Instructions) {
                    var specification = instruction.Operand as MethodSpec;
                    if (specification == null) continue;
                    var target = specification.Method.ResolveMethodDef();
                    if (target != cast && (asRef == null || target != asRef)) continue;
                    var arguments = specification.GenericInstMethodSig.GenericArguments;
                    if (instruction.OpCode != OpCodes.Call || arguments.Count != (target == cast ? 1 : 2) ||
                        arguments.Any(a => a.IsValueType || (target == asRef && a.ElementType != ElementType.Object &&
                            a.ElementType != ElementType.String && a.ElementType != ElementType.Class && !(a is GenericInstSig))))
                        throw new InvalidDataException("Invalid unchecked reference transfer.");
                    instruction.OpCode = OpCodes.Nop; instruction.Operand = null;
                }
            module.Types.Remove(helper);
        }
        return helpers.Length != 0;
    }

    static bool RestoreMemberNames(ModuleDef module, bool methods) {
        string kind = methods ? "method" : "field";
        string prefix = "dnSpy.metadata." + kind + "-name.v1|";
        var maps = new Dictionary<string, UTF8String>(StringComparer.Ordinal);
        var markers = new List<CustomAttribute>();
        foreach (var attribute in module.Assembly.CustomAttributes) {
            if (attribute.TypeFullName != "System.Reflection.ObfuscationAttribute") continue;
            var feature = attribute.NamedArguments.FirstOrDefault(a => a.Name == "Feature");
            string text = feature == null || feature.Value == null ? null : feature.Value.ToString();
            if (text == null || !text.StartsWith(prefix, StringComparison.Ordinal)) continue;
            var parts = text.Substring(prefix.Length).Split('|');
            if (parts.Length != 6) throw new InvalidDataException("Invalid dnSpy " + kind + " name instruction.");
            string key = string.Join("|", parts.Take(5));
            var name = new UTF8String(Convert.FromBase64String(parts[5]));
            UTF8String previous;
            if (maps.TryGetValue(key, out previous) && previous != name)
                throw new InvalidDataException("Conflicting " + kind + " name instructions for one assembly context.");
            maps[key] = name;
            markers.Add(attribute);
        }
        if (markers.Count == 0) return false;
        Func<ITypeDefOrRef, string, string> keyFor = (owner, name) => {
            var specification = owner as TypeSpec;
            if (specification != null) {
                var generic = specification.TypeSig.RemovePinnedAndModifiers() as GenericInstSig;
                if (generic == null) return null;
                owner = generic.GenericType.TypeDefOrRef;
            }
            var assembly = owner.DefinitionAssembly;
            if (assembly == null) return null;
            Func<string, string> encode = value => Convert.ToBase64String(System.Text.Encoding.UTF8.GetBytes(value));
            return string.Join("|", new[] { owner.FullName, assembly.Name.String, assembly.Version.ToString(), assembly.Culture.String ?? "", name }.Select(encode));
        };
        Func<MemberRef, bool> matches = reference => methods ? reference.IsMethodRef : reference.IsFieldRef;
        var references = new HashSet<MemberRef>(module.GetMemberRefs().Where(matches));
        foreach (var method in module.GetTypes().SelectMany(t => t.Methods).Where(m => m.HasBody))
            foreach (var instruction in method.Body.Instructions) {
                var reference = instruction.Operand as MemberRef;
                if (reference == null && instruction.Operand is MethodSpec specification) reference = specification.Method as MemberRef;
                if (reference != null && matches(reference)) references.Add(reference);
            }
        // Capture source keys before mutating definitions; direct definition
        // operands and method specifications follow the same storage/body.
        var definitions = new Dictionary<IMemberDef, UTF8String>();
        var memberNames = new Dictionary<MemberRef, UTF8String>();
        var members = module.GetTypes().SelectMany(t => methods ? t.Methods.Cast<IMemberDef>() : t.Fields.Cast<IMemberDef>());
        foreach (var member in members) {
            var key = keyFor(member.DeclaringType, member.Name);
            UTF8String name;
            if (key != null && maps.TryGetValue(key, out name)) definitions.Add(member, name);
        }
        foreach (var reference in references) {
            var key = keyFor(reference.DeclaringType, reference.Name);
            UTF8String name;
            if (key != null && maps.TryGetValue(key, out name)) memberNames.Add(reference, name);
        }
        if (methods) {
            AvoidGeneratedMethodCollisions(module, definitions, references, memberNames);
            RestoreStateMachineTypeNames(module, definitions);
            foreach (var method in module.GetTypes().SelectMany(t => t.Methods))
                foreach (var marker in method.CustomAttributes.Where(IsOriginalDuplicateMarker).ToArray()) method.CustomAttributes.Remove(marker);
        }
        foreach (var entry in memberNames) entry.Key.Name = entry.Value;
        foreach (var entry in definitions) {
            if (entry.Key is FieldDef field) field.Name = entry.Value;
            else ((MethodDef)entry.Key).Name = entry.Value;
        }
        foreach (var marker in markers) module.Assembly.CustomAttributes.Remove(marker);
        return true;
    }

    static void RestoreStateMachineTypeNames(ModuleDef module, Dictionary<IMemberDef, UTF8String> methodNames) {
        var names = new Dictionary<TypeDef, UTF8String>();
        foreach (var entry in methodNames) {
            var method = entry.Key as MethodDef;
            if (method == null || method.Name == entry.Value) continue;
            foreach (var attribute in method.CustomAttributes) {
                if (attribute.TypeFullName != "System.Runtime.CompilerServices.AsyncStateMachineAttribute" &&
                    attribute.TypeFullName != "System.Runtime.CompilerServices.IteratorStateMachineAttribute" &&
                    attribute.TypeFullName != "System.Runtime.CompilerServices.AsyncIteratorStateMachineAttribute") continue;
                if (attribute.ConstructorArguments.Count != 1) continue;
                var signature = attribute.ConstructorArguments[0].Value as TypeSig;
                var state = signature == null ? null : signature.ToTypeDefOrRef().ResolveTypeDef();
                if (state == null || state.Module != module || state.DeclaringType != method.DeclaringType ||
                    !state.CustomAttributes.Any(a => a.TypeFullName == "System.Runtime.CompilerServices.CompilerGeneratedAttribute") ||
                    !state.Name.String.StartsWith("<" + method.Name.String + ">d__", StringComparison.Ordinal)) continue;
                // Retain the compiler ordinal/arity and the original method's raw
                // UTF8 bytes. References and state-machine attributes share the type.
                var name = new UTF8String(new byte[] { (byte)'<' }.Concat(entry.Value.Data)
                    .Concat(state.Name.Data.Skip(1 + method.Name.Data.Length)).ToArray());
                UTF8String previous;
                if (names.TryGetValue(state, out previous) && previous != name)
                    throw new InvalidDataException("Conflicting state-machine owner names.");
                names[state] = name;
            }
        }
        foreach (var entry in names) {
            foreach (var other in entry.Key.DeclaringType.NestedTypes) {
                UTF8String planned;
                if (other != entry.Key && (names.TryGetValue(other, out planned) ? planned : other.Name) == entry.Value)
                    throw new InvalidDataException("State-machine name restoration would create duplicate nested types.");
            }
        }
        var references = new Dictionary<TypeRef, UTF8String>();
        foreach (var reference in module.GetTypeRefs()) {
            if (reference.DefinitionAssembly == null || reference.DefinitionAssembly.FullName != module.Assembly.FullName) continue;
            foreach (var entry in names)
                if (reference.FullName == entry.Key.FullName) references.Add(reference, entry.Value);
        }
        foreach (var entry in references) entry.Key.Name = entry.Value;
        foreach (var entry in names) entry.Key.Name = entry.Value;
    }

    static bool IsOriginalDuplicateMarker(CustomAttribute attribute) => attribute.TypeFullName == "System.Reflection.ObfuscationAttribute" &&
        attribute.NamedArguments.Any(a => a.Name == "Feature" && a.Value != null && a.Value.ToString() == "dnSpy.metadata.original-duplicate-method.v1");

    static void AvoidGeneratedMethodCollisions(ModuleDef module, Dictionary<IMemberDef, UTF8String> names,
        HashSet<MemberRef> references, Dictionary<MemberRef, UTF8String> referenceNames) {
        var retained = names.Keys.OfType<MethodDef>().ToArray();
        var redirects = new Dictionary<MethodDef, MethodDef>();
        var used = new HashSet<string>(module.GetTypes().SelectMany(t => t.Methods.Select(m => m.Name.String)).Concat(names.Values.Select(n => n.String)), StringComparer.Ordinal);
        foreach (var method in retained) {
            var originalName = names[method];
            foreach (var other in method.DeclaringType.Methods.Where(m => m != method && !redirects.ContainsKey(m)).ToArray()) {
                UTF8String planned;
                if (!names.TryGetValue(other, out planned)) planned = other.Name;
                if (planned != originalName || !new SigComparer().Equals(other.MethodSig, method.MethodSig)) continue;
                if (names.ContainsKey(other) && method.IsPrivate && other.IsPrivate && !method.IsVirtual && !other.IsVirtual &&
                    !method.HasOverrides && !other.HasOverrides && method.CustomAttributes.Any(IsOriginalDuplicateMarker) && other.CustomAttributes.Any(IsOriginalDuplicateMarker)) continue;
                if (names.ContainsKey(other) || !(IsGeneratedType(other.DeclaringType) ||
                    other.CustomAttributes.Any(a => a.TypeFullName == "System.Runtime.CompilerServices.CompilerGeneratedAttribute")))
                    throw new InvalidDataException("Method name restoration would create duplicate runtime methods.");
                // A C# record always synthesizes this method. Keep the retained
                // input implementation and its override family, including custom
                // copy-constructor behavior, rather than add a second clone slot.
                if (originalName == "<Clone>$" && method.IsVirtual && other.IsVirtual && method.MethodSig.Params.Count == 0) {
                    redirects.Add(other, method);
                    continue;
                }
                if (other.IsVirtual || other.HasOverrides)
                    throw new InvalidDataException("Unsupported generated virtual method name collision.");
                string stem = "__DnSpyGeneratedMethod_" + other.MDToken.Raw.ToString("X8"), alias = stem;
                for (int suffix = 1; !used.Add(alias); suffix++) alias = stem + "_" + suffix;
                var name = new UTF8String(alias);
                foreach (var reference in references)
                    if (reference.ResolveMethodDef() == other) referenceNames[reference] = name;
                names.Add(other, name);
            }
        }
        if (redirects.Count == 0) return;
        Func<IMethodDefOrRef, IMethodDefOrRef> redirect = reference => {
            MethodDef replacement;
            return reference is MethodDef definition && redirects.TryGetValue(definition, out replacement) ? replacement : reference;
        };
        foreach (var method in module.GetTypes().SelectMany(t => t.Methods)) {
            if (method.HasBody) foreach (var instruction in method.Body.Instructions) {
                if (instruction.Operand is MethodSpec specification) specification.Method = redirect(specification.Method);
                else if (instruction.Operand is MethodDef definition) instruction.Operand = redirect(definition);
            }
            for (int index = 0; index < method.Overrides.Count; index++) {
                var item = method.Overrides[index];
                method.Overrides[index] = new MethodOverride(redirect(item.MethodBody), redirect(item.MethodDeclaration));
            }
        }
        foreach (var method in redirects.Keys) method.DeclaringType.Methods.Remove(method);
    }

    static bool RestoreTypeNames(ModuleDef module) {
        const string prefix = "dnSpy.metadata.type-name.v1|";
        var maps = new Dictionary<string, UTF8String>(StringComparer.Ordinal);
        var markers = new List<CustomAttribute>();
        foreach (var attribute in module.Assembly.CustomAttributes) {
            if (attribute.TypeFullName != "System.Reflection.ObfuscationAttribute") continue;
            var feature = attribute.NamedArguments.FirstOrDefault(a => a.Name == "Feature");
            string text = feature == null || feature.Value == null ? null : feature.Value.ToString();
            if (text == null || !text.StartsWith(prefix, StringComparison.Ordinal)) continue;
            var parts = text.Substring(prefix.Length).Split('|');
            if (parts.Length != 5) throw new InvalidDataException("Invalid dnSpy type name instruction.");
            string key = string.Join("|", parts.Take(4));
            var name = new UTF8String(Convert.FromBase64String(parts[4]));
            UTF8String previous;
            if (maps.TryGetValue(key, out previous) && previous != name)
                throw new InvalidDataException("Conflicting type name instructions for one assembly context.");
            maps[key] = name;
            markers.Add(attribute);
        }
        if (markers.Count == 0) return false;
        // Capture every key before changing an enclosing type. TypeRefs are also
        // used by generic signatures, method bodies, constraints and attributes.
        var renames = new List<KeyValuePair<ITypeDefOrRef, UTF8String>>();
        foreach (var type in module.GetTypes().Cast<ITypeDefOrRef>().Concat(module.GetTypeRefs())) {
            var assembly = type.DefinitionAssembly;
            if (assembly == null) continue;
            Func<string, string> encode = value => Convert.ToBase64String(System.Text.Encoding.UTF8.GetBytes(value));
            string key = string.Join("|", new[] { type.FullName, assembly.Name.String, assembly.Version.ToString(), assembly.Culture.String ?? "" }.Select(encode));
            UTF8String name;
            if (maps.TryGetValue(key, out name)) renames.Add(new KeyValuePair<ITypeDefOrRef, UTF8String>(type, name));
        }
        AvoidGeneratedTypeCollisions(module, renames);
        var originalTypeKeys = module.GetTypes().Cast<ITypeDefOrRef>().Concat(module.GetTypeRefs())
            .Where(t => t.DefinitionAssembly != null).ToDictionary(t => t, t => BamlTypeKey(t.ReflectionFullName, t.DefinitionAssembly));
        foreach (var rename in renames) rename.Key.Name = rename.Value;
        var bamlNames = new Dictionary<string, string>(StringComparer.Ordinal);
        foreach (var type in originalTypeKeys) {
            if (type.Value == BamlTypeKey(type.Key.ReflectionFullName, type.Key.DefinitionAssembly)) continue;
            string previous;
            if (bamlNames.TryGetValue(type.Value, out previous) && previous != type.Key.ReflectionFullName)
                throw new InvalidDataException("Conflicting BAML type restoration names.");
            bamlNames[type.Value] = type.Key.ReflectionFullName;
        }
        RestoreBamlTypeNames(module, bamlNames);
        foreach (var marker in markers) module.Assembly.CustomAttributes.Remove(marker);
        return true;
    }

    static string BamlTypeKey(string name, IAssembly assembly) {
        return name + "|" + new AssemblyNameInfo(assembly.FullName).FullName;
    }

    static bool UsesStandardResourceReader(byte[] data) {
        if (data.Length < 12) return false;
        using (var reader = new BinaryReader(new MemoryStream(data))) {
            if (reader.ReadUInt32() != 0xbeefcace || reader.ReadInt32() != 1) return false;
            int headerLength = reader.ReadInt32();
            if (headerLength < 0 || headerLength > data.Length - 12)
                throw new InvalidDataException("Invalid resource reader header length.");
            string readerType = reader.ReadString();
            if (reader.BaseStream.Position > 12L + headerLength)
                throw new InvalidDataException("Invalid resource reader header.");
            return readerType.Split(',')[0].Trim() == "System.Resources.ResourceReader";
        }
    }

    static void RestoreBamlTypeNames(ModuleDef module, Dictionary<string, string> names) {
        if (names.Count == 0) return;
        for (int index = 0; index < module.Resources.Count; index++) {
            var resource = module.Resources[index] as EmbeddedResource;
            if (resource == null || !resource.Name.String.EndsWith(".resources", StringComparison.OrdinalIgnoreCase)) continue;
            byte[] resourceData = resource.CreateReader().ToArray();
            // WPF BAML uses the standard resource format. Other readers (such
            // as preserialized ResX readers) must retain their original bytes;
            // never load their reader types or deserialize their values here.
            if (!UsesStandardResourceReader(resourceData)) continue;
            var entries = new List<Tuple<string, string, byte[]>>();
            bool changed = false;
            using (var stream = new MemoryStream(resourceData))
            using (var reader = new System.Resources.ResourceReader(stream)) {
                var enumerator = reader.GetEnumerator();
                while (enumerator.MoveNext()) {
                    string key = (string)enumerator.Key, type;
                    byte[] data;
                    // Never deserialize arbitrary resource values.
                    reader.GetResourceData(key, out type, out data);
                    if (key.EndsWith(".baml", StringComparison.OrdinalIgnoreCase) &&
                        (type == "ResourceTypeCode.Stream" || type == "ResourceTypeCode.ByteArray")) {
                        if (data.Length < 4 || BitConverter.ToInt32(data, 0) != data.Length - 4)
                            throw new InvalidDataException("Invalid BAML resource stream length: " + key);
                        BamlDocument document;
                        using (var baml = new MemoryStream(data, 4, data.Length - 4))
                            document = BamlReader.ReadDocument(baml, System.Threading.CancellationToken.None);
                        var assemblies = document.OfType<AssemblyInfoRecord>().ToDictionary(a => (int)a.AssemblyId, a => new AssemblyNameInfo(a.AssemblyFullName));
                        bool renamed = false;
                        foreach (var record in document.OfType<TypeInfoRecord>()) {
                            AssemblyNameInfo assembly;
                            if (record.AssemblyId == ushort.MaxValue) assembly = new AssemblyNameInfo(module.Assembly.FullName);
                            else if (!assemblies.TryGetValue(record.AssemblyId & 0xfff, out assembly))
                                throw new InvalidDataException("Missing BAML assembly record: " + key);
                            string name;
                            if (names.TryGetValue(BamlTypeKey(record.TypeFullName, assembly), out name)) {
                                record.TypeFullName = name;
                                renamed = true;
                            }
                        }
                        if (renamed) {
                            using (var baml = new MemoryStream()) {
                                BamlWriter.WriteDocument(document, baml);
                                byte[] payload = baml.ToArray();
                                data = new byte[payload.Length + 4];
                                Buffer.BlockCopy(BitConverter.GetBytes(payload.Length), 0, data, 0, 4);
                                Buffer.BlockCopy(payload, 0, data, 4, payload.Length);
                            }
                            changed = true;
                        }
                    }
                    entries.Add(Tuple.Create(key, type, data));
                }
            }
            if (!changed) continue;
            using (var stream = new MemoryStream()) {
                using (var writer = new System.Resources.ResourceWriter(stream)) {
                    foreach (var entry in entries) writer.AddResourceData(entry.Item1, entry.Item2, entry.Item3);
                    writer.Generate();
                    module.Resources[index] = new EmbeddedResource(resource.Name, stream.ToArray(), resource.Attributes);
                }
            }
        }
    }

    static string PlannedTypeKey(ITypeDefOrRef type, Dictionary<TypeDef, UTF8String> names) {
        UTF8String name;
        if (!(type is TypeDef) || !names.TryGetValue((TypeDef)type, out name)) name = type.Name;
        var parent = type.DeclaringType;
        var ns = (type as TypeDef)?.Namespace ?? (type as TypeRef)?.Namespace ?? UTF8String.Empty;
        return (parent == null ? Convert.ToBase64String(ns.Data) + "." : PlannedTypeKey(parent, names) + "/") + Convert.ToBase64String(name.Data);
    }

    static bool IsGeneratedType(TypeDef type) {
        for (; type != null; type = type.DeclaringType)
            if (type.CustomAttributes.Any(a => a.TypeFullName == "System.Runtime.CompilerServices.CompilerGeneratedAttribute")) return true;
        return false;
    }

    static void AvoidGeneratedTypeCollisions(ModuleDef module, List<KeyValuePair<ITypeDefOrRef, UTF8String>> renames) {
        var types = module.GetTypes().ToArray();
        var names = renames.Where(p => p.Key is TypeDef).ToDictionary(p => (TypeDef)p.Key, p => p.Value);
        var originalInstructions = new HashSet<TypeDef>(names.Keys);
        var generated = new Dictionary<TypeDef, UTF8String>();
        var used = new HashSet<string>(types.Select(t => t.Name.String).Concat(names.Values.Select(n => n.String)), StringComparer.Ordinal);
        for (;;) {
            var collision = types.GroupBy(t => PlannedTypeKey(t, names)).FirstOrDefault(g => g.Count() > 1);
            if (collision == null) break;
            // A retained input helper owns its original runtime name. A new C#
            // helper may happen to get that same name; move only the new helper.
            var retained = collision.Where(originalInstructions.Contains).ToArray();
            var disposable = collision.Where(t => !originalInstructions.Contains(t)).ToArray();
            if (retained.Length != 1 || disposable.Length == 0 || disposable.Any(t => !IsGeneratedType(t)))
                throw new InvalidDataException("Type name restoration would create duplicate runtime types.");
            foreach (var type in disposable) {
                string stem = "__DnSpyGenerated_" + type.MDToken.Raw.ToString("X8"), alias = stem;
                for (int suffix = 1; !used.Add(alias); suffix++) alias = stem + "_" + suffix;
                names[type] = generated[type] = new UTF8String(alias);
            }
        }
        if (generated.Count == 0) return;
        var empty = new Dictionary<TypeDef, UTF8String>();
        var oldNames = generated.ToDictionary(p => PlannedTypeKey(p.Key, empty), p => p.Value);
        foreach (var reference in module.GetTypeRefs()) {
            var assembly = reference.DefinitionAssembly;
            if (assembly == null || assembly.Name != module.Assembly.Name || assembly.Version != module.Assembly.Version || assembly.Culture != module.Assembly.Culture) continue;
            UTF8String name;
            if (oldNames.TryGetValue(PlannedTypeKey(reference, empty), out name)) renames.Add(new KeyValuePair<ITypeDefOrRef, UTF8String>(reference, name));
        }
        renames.AddRange(generated.Select(p => new KeyValuePair<ITypeDefOrRef, UTF8String>(p.Key, p.Value)));
    }

    static string TypeKey(ITypeDefOrRef type) { return type.FullName + ", " + type.DefinitionAssembly.FullName; }
    static bool RestoreDelegateMethods(ModuleDef module) {
        var attributes = module.Types.Where(t => t.CustomAttributes.Any(a => a.TypeFullName == "System.Reflection.ObfuscationAttribute" &&
            a.NamedArguments.Any(n => n.Name == "Feature" && n.Value != null && n.Value.ToString() == "dnSpy.metadata.delegate-methods.v1"))).ToArray();
        if (attributes.Length == 0) return false;
        var maps = new Dictionary<string, ITypeDefOrRef>(StringComparer.Ordinal);
        foreach (var marker in attributes) {
            foreach (var attribute in module.Assembly.CustomAttributes.Where(a => a.AttributeType.ResolveTypeDef() == marker).ToArray()) {
                if (attribute.ConstructorArguments.Count != 2) throw new InvalidDataException("Invalid delegate method mapping.");
                var source = (attribute.ConstructorArguments[0].Value as TypeSig).ToTypeDefOrRef();
                var target = (attribute.ConstructorArguments[1].Value as TypeSig).ToTypeDefOrRef();
                if (source == null || target == null) throw new InvalidDataException("Missing delegate method mapping types.");
                maps.Add(TypeKey(source), target);
                module.Assembly.CustomAttributes.Remove(attribute);
            }
        }
        // Load instruction operands before changing ownership: dnlib metadata rows are lazy.
        var methods = module.GetTypes().SelectMany(t => t.Methods).ToArray();
        var references = new HashSet<MemberRef>(module.GetMemberRefs());
        foreach (var method in methods.Where(m => m.HasBody)) foreach (var instruction in method.Body.Instructions) {
            var reference = instruction.Operand as MemberRef;
            var specification = instruction.Operand as MethodSpec;
            if (reference == null && specification != null) reference = specification.Method as MemberRef;
            if (reference != null) references.Add(reference);
        }
        foreach (var reference in references) {
            var declaring = reference.DeclaringType;
            if (declaring == null) continue;
            var generic = (declaring as TypeSpec)?.TypeSig as GenericInstSig;
            var definition = generic == null ? declaring : generic.GenericType.TypeDefOrRef;
            ITypeDefOrRef target;
            if (!maps.TryGetValue(TypeKey(definition), out target)) continue;
            reference.Class = generic == null ? (IMemberRefParent)target : new TypeSpecUser(new GenericInstSig(new ClassSig(target), generic.GenericArguments.ToArray()));
        }
        foreach (var companion in module.GetTypes().ToArray()) {
            ITypeDefOrRef destination;
            if (!maps.TryGetValue(TypeKey(companion), out destination)) continue;
            var target = destination.ResolveTypeDef();
            if (target == null || target.Module != module || target.BaseType == null || target.BaseType.FullName != "System.MulticastDelegate")
                throw new InvalidDataException("Delegate method destination is not a local delegate.");
            if (!companion.IsAbstract || !companion.IsSealed || companion.Methods.Any(m => !m.IsStatic))
                throw new InvalidDataException("Invalid delegate method companion.");
            foreach (var method in companion.Methods.ToArray()) { companion.Methods.Remove(method); target.Methods.Add(method); }
            foreach (var field in companion.Fields.ToArray()) { companion.Fields.Remove(field); target.Fields.Add(field); }
            foreach (var nested in companion.NestedTypes.ToArray()) { companion.NestedTypes.Remove(nested); target.NestedTypes.Add(nested); }
            if (companion.DeclaringType == null) module.Types.Remove(companion); else companion.DeclaringType.NestedTypes.Remove(companion);
        }
        foreach (var marker in attributes) module.Types.Remove(marker);
        return true;
    }
