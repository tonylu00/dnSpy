using System;
using System.Runtime.CompilerServices;
using System.Threading.Tasks;

public class Awaitable {
    readonly Task<int> task;
    public static int FactoryCalls;
    public Awaitable(Task<int> task) { this.task = task; }
    public TaskAwaiter<int> GetAwaiter() { FactoryCalls++; return task.GetAwaiter(); }
}

public static class Fixture {
    [AsyncStateMachine(typeof(DetachedState))]
    public static Task<int> ReadDetached(Awaitable input) {
        var machine = new DetachedState();
        machine.builder = AsyncTaskMethodBuilder<int>.Create();
        machine.input = input;
        machine.state = -1;
        machine.builder.Start(ref machine);
        return machine.builder.Task;
    }
    public static async Task<int> ReadCanonical(Awaitable input) { return await input; }

                goto Success;

                goto Success;
            Suspend:

                saved = awaiter;

                return;
            Success: ;
            }
            catch (Exception error) { state = -2; builder.SetException(error); return; }
            state = -2;
            builder.SetResult(result);
        }
